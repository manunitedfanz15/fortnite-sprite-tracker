<script>
const KEY="fortniteSpriteTracker35";
const base=[
 ["Halfpeel'd","Rare","halfpeeld",3],
 ["Re-Pete","Rare","re-pete",3],
 ["Barry O'Fish","Rare","barry-o-fish",3],
 ["Wayfinder","Rare","wayfinder",3],
 ["Kelper","Epic","kelper",3],
 ["Exmarks","Epic","exmarks",3],
 ["Lil' Squibbly","Epic","lil-squibbly",3],
 ["Chester","Epic","chester",3],
 ["Lil' Flacko","Legendary","lil-flacko",3],
 ["Transponder Snail","Legendary","transponder-snail",1],
 ["Sweet Dreams","Mythic","sweet-dreams",1],
 ["Plundarrgh","Mythic","plundarrgh",3],
 ["Lil' Uzi Vert Jr.","Mythic","lil-uzi-vert-jr",3]
];
const variants=["Normal","Fortune","Cursed"];
const sprites=base.flatMap((b,i)=>variants.slice(0,b[3]).map(v=>({
 id:i+"-"+v,name:b[0],rarity:b[1],slug:b[2],variant:v,
 image:"https://www.fortnitetools.com/images/sprites/"+
 (v==="Normal"?"":v.toLowerCase()+"-")+b[2]+"-sprite.webp"
})));
const labels={owned:"Owned",mastered:"Mastered",found:"Found in Island",won:"Won With"};
let data={};
try{data=JSON.parse(localStorage.getItem(KEY)||"{}")||{};}catch(e){data={};}
let tab="overview",view="all";
const $=id=>document.getElementById(id);

function state(s){if(!data[s.id])data[s.id]={};return data[s.id];}
function save(){localStorage.setItem(KEY,JSON.stringify(data));updateStats();}
function complete(s){
 const d=state(s);
 return tab==="overview"
  ?Object.keys(labels).every(k=>!!d[k])
  :!!d[tab];
}
function getList(){
 const q=$("search").value.toLowerCase().trim();
 return sprites.filter(s=>{
  if(q&&!(s.name+" "+s.variant+" "+s.rarity).toLowerCase().includes(q))return false;
  if($("rarity").value!=="all"&&s.rarity!==$("rarity").value)return false;
  if($("variant").value!=="all"&&s.variant!==$("variant").value)return false;
  if(view==="remaining"&&complete(s))return false;
  if(view==="completed"&&!complete(s))return false;
  return true;
 });
}
function card(s){
 const d=state(s);
 const img=`<img class="sprite-img" src="${s.image}" alt="${s.name} ${s.variant}" loading="lazy" onerror="this.style.display='none';this.nextElementSibling.style.display='flex'">`;
 const fallback=`<div class="sprite-img" style="display:none;align-items:center;justify-content:center;text-align:center;padding:7px;font-size:11px;color:#aebbd1">Image unavailable</div>`;
 let checks="";
 if(tab==="overview"){
  checks='<div class="checks">'+Object.keys(labels).map(k=>`
   <label class="check ${d[k]?'done':''}">
    <input type="checkbox" data-id="${s.id}" data-key="${k}" ${d[k]?'checked':''}>
    <span>${labels[k]}</span>
   </label>`).join("")+'</div>';
 }else{
  const k=tab;
  checks=`<div class="single"><label class="check ${d[k]?'done':''}">
   <input type="checkbox" data-id="${s.id}" data-key="${k}" ${d[k]?'checked':''}>
   <span>${d[k]?"Completed — click to undo":"Mark as "+labels[k]}</span>
  </label></div>`;
 }
 return `<article class="card">
  <div class="sprite-top">${img}${fallback}<div>
   <div class="name">${s.name}</div>
   <div class="meta">${s.variant} Variant</div>
   <span class="rarity ${s.rarity.toLowerCase()}">${s.rarity}</span>
  </div></div>${checks}
 </article>`;
}
function updateStats(){
 const total=sprites.length;
 $("total").textContent=total;
 Object.keys(labels).forEach(k=>{
  const n=sprites.filter(s=>data[s.id]&&data[s.id][k]).length;
  $(k+"Count").textContent=n+"/"+total;
  $(k+"Bar").style.width=(n/total*100)+"%";
 });
}
function render(){
 const list=getList();
 $("grid").innerHTML=list.length?list.map(card).join(""):'<div class="empty">No Sprites match your filters.</div>';
 $("views").style.display=tab==="overview"?"none":"flex";
 $("note").textContent=tab==="overview"
  ?"Overview: all Sprites and all four tracking checkboxes."
  :labels[tab]+" — "+view.charAt(0).toUpperCase()+view.slice(1);
 document.querySelectorAll("[data-tab]").forEach(b=>b.classList.toggle("active",b.dataset.tab===tab));
 document.querySelectorAll("[data-view]").forEach(b=>b.classList.toggle("active",b.dataset.view===view));
 updateStats();
}
$("tabs").onclick=e=>{
 const b=e.target.closest("[data-tab]");if(!b)return;
 tab=b.dataset.tab;view="all";render();
};
$("views").onclick=e=>{
 const b=e.target.closest("[data-view]");if(!b)return;
 view=b.dataset.view;render();
};
$("grid").onchange=e=>{
 const c=e.target.closest('input[type="checkbox"]');if(!c)return;
 state({id:c.dataset.id})[c.dataset.key]=c.checked;
 save();render();
};
["search","rarity","variant"].forEach(id=>$(id).addEventListener(id==="search"?"input":"change",render));
$("export").onclick=()=>{
 const blob=new Blob([JSON.stringify({version:1,data},null,2)],{type:"application/json"});
 const a=document.createElement("a");a.href=URL.createObjectURL(blob);
 a.download="fortnite-sprite-tracker-backup.json";a.click();
 URL.revokeObjectURL(a.href);
};
$("import").onclick=()=>$("file").click();
$("file").onchange=async e=>{
 const f=e.target.files[0];if(!f)return;
 try{
  const parsed=JSON.parse(await f.text()),incoming=parsed.data||parsed;
  if(!incoming||typeof incoming!=="object")throw Error();
  data={...data,...incoming};save();render();alert("Backup imported!");
 }catch(err){alert("Could not read that backup file.");}
 e.target.value="";
};
$("reset").onclick=()=>{
 if(!confirm("Reset all Sprite progress? Export a backup first if needed."))return;
 data={};save();render();
};
render();
</script>
